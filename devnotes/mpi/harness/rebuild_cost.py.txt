def status(key):
    with open("/proc/self/status") as f:
        for line in f:
            if line.startswith(key):
                return int(line.split()[1]) / 1024.0


records = []  # (rebuild index, label, seconds)
rebuilds = []  # (kind, seconds, rss before, peak above before)
current = [None]


def timed(owner, name, label):
    original = getattr(owner, name)

    @functools.wraps(original)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        try:
            return original(*args, **kwargs)
        finally:
            records.append((current[0], label, time.perf_counter() - start))

    setattr(owner, name, wrapper)


def timedRebuild():
    original = ned.NED.buildEquationSystem

    @functools.wraps(original)
    def wrapper(self, model, step, previous=None):
        kind = "carried" if previous is not None else ("start" if not rebuilds else "afresh")
        current[0] = len(rebuilds)
        rebuilds.append(None)
        rssBefore = status("VmRSS")
        with open("/proc/self/clear_refs", "w") as f:
            f.write("5")
        start = time.perf_counter()
        try:
            return original(self, model, step, previous)
        finally:
            rebuilds[current[0]] = (kind, time.perf_counter() - start, rssBefore, status("VmHWM") - rssBefore)
            current[0] = None

    ned.NED.buildEquationSystem = wrapper
