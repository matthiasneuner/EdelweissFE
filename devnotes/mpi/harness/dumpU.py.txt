"""Run test decks in-process and save the final solution bit-exactly: dumpU.py OUTDIR DIR1 DIR2 ..."""
import os, sys, io, contextlib
import numpy as np
from edelweissfe.drivers.inputfiledrivensimulation import finiteElementSimulation
from edelweissfe.utils.inputfileparser import parseInputFile
out = os.path.abspath(sys.argv[1]); os.makedirs(out, exist_ok=True)
rank = 0
try:
    from edelweissfe.domaindecomposition.mpienvironment import processRank
    rank = processRank()
except ImportError:
    pass
for d in sys.argv[2:]:
    d = os.path.abspath(d); cwd = os.getcwd(); os.chdir(d)
    try:
        inp = parseInputFile("test.inp")
        with contextlib.redirect_stdout(io.StringIO()):
            model, _ = finiteElementSimulation(inp, verbose=False, suppressPlots=True)
        U = np.hstack([f["U"].flatten() for f in model.nodeFields.values()] + [[v.value for v in model.scalarVariables.values()]]).flatten()
        if rank == 0:
            np.save(os.path.join(out, os.path.basename(os.path.dirname(d)) + "__" + os.path.basename(d) + ".npy"), U)
            print(os.path.basename(d), "OK", U.size)
    except Exception as e:
        print(os.path.basename(d), "ERROR", repr(e)[:300])
    finally:
        os.chdir(cwd)
